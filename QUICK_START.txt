📤 STEP 1: PUSH FROM LOCAL (Windows)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
cd c:\Users\HANIMA\OneDrive\Desktop\new_project_algobiz\HRC_CRM
git add .
git commit -m "Add Celery, email reminders, day-based dates"
git push origin main


🖥️ STEP 2: DEPLOY ON SERVER (EC2)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ssh -i your-key.pem ubuntu@your-server-ip
cd /var/www/hrc-crm-backend
git pull origin main
source venv/bin/activate
pip install -r requirements.txt
python manage.py migrate
python manage.py collectstatic --noinput
sudo systemctl restart hrc-crm-backend


🔧 STEP 3: INSTALL REDIS (First Time Only)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
sudo apt update
sudo apt install redis-server -y
sudo systemctl start redis
sudo systemctl enable redis
redis-cli ping  # Should return: PONG


🔄 STEP 4: SETUP CELERY (First Time Only)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
See PUSH_TO_SERVER.md for detailed Celery setup
Or run the provided systemd service files


✅ STEP 5: VERIFY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

sudo systemctl status celery-worker
sudo systemctl status celery-beat
sudo systemctl status redis


📧 STEP 6: SCHEDULE EMAILS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Go to: https://crm.hrccosmos.com/admin/
Navigate to: Periodic Tasks
Add new task:
  - Name: Send Plan Expiry Reminders
  - Task: user.tasks.send_expiry_reminders
  - Interval: Every 1 day
  - Enabled: ✅
